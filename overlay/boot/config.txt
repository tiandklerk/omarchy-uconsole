# ---------------------------------------------------------------------------
# omarchy-uconsole - Raspberry Pi firmware configuration
# Target: ClockworkPi uConsole carrier board + Raspberry Pi Compute Module 5
# ---------------------------------------------------------------------------
# The values here are derived from ak-rex's ClockworkPi pi-gen boot files,
# adapted for Arch Linux ARM (which loads the kernel by name from this file
# rather than relying on the default kernel8.img).
# ---------------------------------------------------------------------------

arm_64bit=1

# Deliberately the stock name. ak-rex's working CM5 images ship kernel8.img and
# let the firmware find it; a custom kernel= filename is one more thing that can
# differ from a known-good reference for no benefit.
kernel=kernel8.img

# Serial console on the UART pins. Costs nothing when unused and is the only
# way to see firmware and early-kernel output when the panel is dark.
enable_uart=1

# The uConsole has no HDMI-attached display at boot and its DSI panel is not a
# standard "LCD"; let the overlay describe it instead of the firmware probing.
ignore_lcd=1
disable_overscan=1
max_framebuffers=2

# On-board audio. The uConsole's speaker amp is wired to GPIO 12/13 expecting
# PWM audio, but the two Compute Modules produce it completely differently, so
# the overlay lives in the per-model sections below:
#
#   CM4 (BCM2711): the VPU converts PCM to PWM        -> audremap
#   CM5 (BCM2712): no VPU path at all; an RP1 hardware
#                  block does the conversion          -> audremap-pi5
#
# Using plain `audremap` on a CM5 is silently inert: the stream plays into the
# RP1's spdif-dit dummy codec and nothing reaches the speaker. Needs kernel
# >= 6.12.17 for the RP1 audio-out driver.
dtparam=audio=on

# uConsole's USB is host-only on the internal hub.
dtoverlay=dwc2,dr_mode=host

# Use the external antenna connector fitted to the uConsole shell, not the
# CM5's on-board antenna. Without this, wifi range is very poor.
dtparam=ant2

[cm5]
dtoverlay=audremap-pi5,pins_12_13

# Speaker amplifier enable. VERIFIED ON HARDWARE: without this the whole audio
# chain is correct and completely silent - PipeWire routes the stream, the RP1
# block converts it, and the amp is simply powered down. ak-rex sets the same
# line for the CM3+/CM4s variants; the CM5 needs it too.
# GPIO11 is gpio-580 in the kernel's numbering, which is the "GPIO 580" the
# Raspberry Pi forum thread refers to.
gpio=11=op,dh

# The GPU. THIS LINE IS LOAD-BEARING: without it the v3d render node never
# appears, Mesa falls back to llvmpipe, EGL fails with "failed to create dri2
# screen", and the compositor draws nothing onto a perfectly lit panel. The
# uConsole overlay below brings up the DSI *display*; the GPU is separate.
# ak-rex's config carries this under a [pi5] section - dropping that section
# when writing a CM5-specific config is exactly how this gets lost.
dtoverlay=vc4-kms-v3d-pi5

# Brings up: the 480x1280 DSI panel, the OCP8178 backlight, the ClockworkPi
# matrix keyboard/trackball, the AXP228 PMIC (battery + charging), and the
# audio codec. Provided by ak-rex's kernel tree - not present in mainline or
# in the stock Raspberry Pi kernel.
dtoverlay=clockworkpi-uconsole-cm5

[cm4]
dtoverlay=audremap,pins_12_13
dtoverlay=clockworkpi-uconsole-cm4

[all]
